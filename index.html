function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetAbsensi = ss.getSheetByName("Absensi") || ss.insertSheet("Absensi");
    var sheetSiswa = ss.getSheetByName("Siswa") || ss.insertSheet("Siswa");
    
    // Header Sheet 1: Absensi (A1:H1)
    if (sheetAbsensi.getLastRow() === 0) {
      sheetAbsensi.appendRow(["Timestamp", "Tanggal", "Jam", "NISN", "Nama", "Kelas", "Jenis Absen", "Status"]);
    }
    
    // Header Sheet 2: Siswa (A1:D1)
    if (sheetSiswa.getLastRow() === 0) {
      sheetSiswa.appendRow(["NISN", "Nama", "Kelas", "Gender"]);
    }
    
    var data = JSON.parse(e.postData.contents);
    
    if (data.action === "register_student") {
      sheetSiswa.appendRow([
        "'" + (data.nisn || ''),
        data.nama || '',
        data.kelas || '',
        data.gender || ''
      ]);
    } else {
      var now = new Date();
      sheetAbsensi.appendRow([
        data.timestamp || now.toLocaleString('id-ID'),
        data.tanggal || now.toISOString().split('T')[0],
        data.waktu || now.toLocaleTimeString('id-ID'),
        "'" + (data.nisn || ''),
        data.nama || '',
        data.kelas || '',
        data.tipe || 'Masuk',
        data.status || 'TEPAT WAKTU'
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
