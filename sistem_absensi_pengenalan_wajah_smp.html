Integrasi Google Spreadsheet Web App

Masukkan Web App URL yang didapatkan dari Google Apps Script Deployment.

Kode Google Apps Script (GAS) Backend

Salin kode di bawah ini ke Google Sheets: Extensions > Apps Script.

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Header
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID", "Tanggal", "Waktu", "NISN", "Nama Siswa", "Kelas", "Tipe", "Status"]);
    }
    
    sheet.appendRow([
      data.id || Date.now(),
      data.tanggal || new Date().toISOString().split('T')[0],
      data.waktu || new Date().toLocaleTimeString('id-ID'),
      "'" + (data.nisn || ''),
      data.nama || '',
      data.kelas || '',
      data.tipe || 'MASUK',
      data.status || 'HADIR'
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
